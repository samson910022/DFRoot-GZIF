set(JNI_DIR ${CMAKE_CURRENT_SOURCE_DIR})

set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} -I${JNI_DIR}")
set(CMAKE_C_FLAGS   "${CMAKE_C_FLAGS} ${C_FLAGS} -I${JNI_DIR} -Wall -Wextra -fno-stack-protector -fomit-frame-pointer")

# Build into the binary directory so generated bytes never live in the source
# tree (they are still embedded into exp.c via DFROOT_SPLICEHELPER_PATH below).
set(SPLICEHELPER "${CMAKE_CURRENT_BINARY_DIR}/splicehelper")
add_custom_command(
    OUTPUT "${SPLICEHELPER}"
    COMMAND "${CMAKE_C_COMPILER}"
            --target=${CMAKE_C_COMPILER_TARGET}
            --sysroot=${CMAKE_SYSROOT}
            "${JNI_DIR}/splicehelper.c"
            -o "${SPLICEHELPER}"
            -nodefaultlibs -nostartfiles -ffreestanding -static
    COMMAND "${CMAKE_STRIP}" "${SPLICEHELPER}"
    DEPENDS "${JNI_DIR}/splicehelper.c"
    COMMENT "Building splicehelper"
)
add_custom_target(splicehelper_bin DEPENDS "${SPLICEHELPER}")
# splicehelper is compiled here from the reviewed splicehelper.c, so no external
# blob/hash pin is needed. Its bytes are NOT guaranteed to be bit-for-bit
# identical across toolchains (compiler, flags and sysroot all matter), so we do
# not claim reproducibility: record_build_provenance.cmake records the sha256 of
# both the built binary and the source instead.

# The .ko below is a gitignored, externally supplied binary embedded by .incbin
# in exp.c and later loaded in ring0. Pin its SHA-256 and verify it at build
# time, before exp.c is compiled (M9 supply-chain hardening).
set(KO_FILE "${JNI_DIR}/ko/dfroot-android13-5.15.ko")
set(KO_FILES
    "${KO_FILE}"
)
# Accepted SHA-256 pins for the embedded dfroot.ko. The first two are from the
# same already-reviewed DDK build and differ only because CI runs an extra
# `llvm-objcopy -R .note.gnu.build-id` after the DDK build:
#   94f8be... = local `make` output from lkm/ (WITH .note.gnu.build-id), 13504 B
#   9dfb8b... = old CI artifact (build-id stripped), 13384 B; byte-for-byte the
#               KO shipped in the v4.8-gzif release APK (obsolete for this head).
#   e10425... = RE-PIN for this head: the CI artifact rebuilt after the UMH
#               typedef/CFI correction in lkm/dfroot.c was shown to have
#               different bytes (13456 B, build-id stripped) by the CI KO
#               policy gate. Added deliberately in a reviewed commit; not
#               auto-generated. The old pins are retained as historical
#               repository records and are not evidence for this head.
# Adding a hash still goes through review; unknown blobs remain fail-closed.
set(KO_EXPECTED_SHA256S
    "94f8be987a2f8bdc13539e42a5c3cb4c0f606024b8b31601496c5d5e5c0025a7"
    "9dfb8b71b306f5a38fd8b35b283d4dab8b4eb25a08ad97ca0f4ef705b6151c68"
    "e10425d055ff6803856cc24f6f4fe2e2e09746df195c50503811356eb87df605"
)
set(KO_VERIFY_SCRIPT "${JNI_DIR}/verify_ko_sha256.cmake")
set(KO_VERIFY_STAMP "${CMAKE_CURRENT_BINARY_DIR}/dfroot-ko.sha256.verified")
set(KO_VERIFY_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/dfroot-ko.sha256.manifest")

# Final packaged-byte gate: verify the bytes ACTUALLY linked into libdfroot.so,
# not just the .ko on disk. objcopy is mandatory for that gate -- fail configure
# rather than silently produce an unchecked artifact. The NDK supplies
# llvm-objcopy as CMAKE_OBJCOPY for this toolchain.
if(NOT CMAKE_OBJCOPY OR "${CMAKE_OBJCOPY}" STREQUAL "")
    message(FATAL_ERROR
        "CMAKE_OBJCOPY is required by the packaged-KO byte gate but is "
        "unavailable in this toolchain. Refusing to configure without it; "
        "never skip the packaged-byte verification.")
endif()
set(PACKAGED_KO_VERIFY_SCRIPT "${JNI_DIR}/verify_packaged_ko.cmake")
# Two-stage artifact chain (see the packaged-KO section below): the linker
# writes an UNVERIFIED .so into a staging directory; only the verified copy at
# the path AGP packages (CMAKE_LIBRARY_OUTPUT_DIRECTORY) is distributable.
# The manifest lives in this CMake build dir (under app/.cxx, where CI finds
# every one) and records the verified copy's path so CI can bind each manifest
# to the exact libdfroot.so AGP packaged.
set(PACKAGED_KO_STAGING_DIR "${CMAKE_CURRENT_BINARY_DIR}/unverified")
set(PACKAGED_KO_RAW_SO "${PACKAGED_KO_STAGING_DIR}/libdfroot.so")
set(PACKAGED_KO_VERIFIED_SO "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}/libdfroot.so")
set(PACKAGED_KO_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/dfroot-packaged-ko.manifest")
# Accepted byte counts of the reviewed artifacts: the old local make output
# (13504 B, WITH .note.gnu.build-id), the old CI artifact (13384 B, build-id
# stripped), and the re-pinned CI artifact for this head (13456 B, stripped).
# These are sizes only; a size can never make an unknown hash pass and no hash
# pin is added or relaxed here.
set(KO_EXPECTED_SIZES
    13504
    13384
    13456
)

# DEPRECATED / UNSUPPORTED: a source-only local experiment that disables just
# the .ko-on-disk pin check before .incbin. It cannot bypass the ALWAYS-RUN
# packaged-KO byte gate below, cannot affect any distributable artifact, and an
# unknown blob may pass this source-only check but still fails the build at the
# final byte gate. Never use it for CI, release or publishing builds.
option(DFROOT_SKIP_KO_HASH_CHECK "DEPRECATED, unsupported source-only local KO pin-check skip (cannot bypass packaged-byte gate)" OFF)
set(DFROOT_SKIP_REASON "" CACHE STRING "Required local justification when DFROOT_SKIP_KO_HASH_CHECK=ON")

# CMake can be driven without CMAKE_BUILD_TYPE (multi-config generators); pass
# an explicit label so the verify script's required argument is always present.
# Policy authority, strongest first: the environment markers (CI,
# GITHUB_ACTIONS, DFROOT_RELEASE) are authoritative, and so is the Android
# variant Gradle passes as -DDFROOT_ANDROID_VARIANT=<debug|release>. Both beat
# CMAKE_BUILD_TYPE, which is read from the cache and can be stale when a cached
# build directory is reused for a different build type -- a stale "Debug" label
# must never weaken the skip guard below (or the one in the verify script).
# CI exports DFROOT_RELEASE=1 for release builds as an additional marker.
if(DEFINED DFROOT_ANDROID_VARIANT AND NOT "${DFROOT_ANDROID_VARIANT}" STREQUAL "")
    string(TOLOWER "${DFROOT_ANDROID_VARIANT}" dfroot_variant_lower)
    if(dfroot_variant_lower STREQUAL "release")
        set(DFROOT_EFFECTIVE_BUILD_TYPE "Release")
    elseif(dfroot_variant_lower STREQUAL "debug")
        set(DFROOT_EFFECTIVE_BUILD_TYPE "Debug")
    else()
        set(DFROOT_EFFECTIVE_BUILD_TYPE "${DFROOT_ANDROID_VARIANT}")
    endif()
elseif(CMAKE_BUILD_TYPE STREQUAL "")
    set(DFROOT_EFFECTIVE_BUILD_TYPE "Unspecified")
else()
    set(DFROOT_EFFECTIVE_BUILD_TYPE "${CMAKE_BUILD_TYPE}")
endif()
if(NOT DEFINED DFROOT_ANDROID_VARIANT OR "${DFROOT_ANDROID_VARIANT}" STREQUAL "")
    set(DFROOT_ANDROID_VARIANT "unspecified")
endif()

if(DFROOT_SKIP_KO_HASH_CHECK)
    # Configure-time guard (the verify script enforces the same policy again so
    # a direct `cmake -P` call cannot bypass it). Fail closed in CI/release:
    # an unverified LKM must never be shipped from an automated build. Local
    # experiments may opt out with a loud warning and a recorded reason.
    if(DEFINED ENV{CI} OR DEFINED ENV{GITHUB_ACTIONS} OR DEFINED ENV{DFROOT_RELEASE})
        message(FATAL_ERROR "DFROOT_SKIP_KO_HASH_CHECK is forbidden in CI/release builds")
    endif()
    string(TOUPPER "${DFROOT_EFFECTIVE_BUILD_TYPE}" dfroot_build_type_upper)
    if(dfroot_build_type_upper STREQUAL "RELEASE")
        message(FATAL_ERROR
            "DFROOT_SKIP_KO_HASH_CHECK is forbidden for Release builds "
            "(CMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}, "
            "DFROOT_ANDROID_VARIANT=${DFROOT_ANDROID_VARIANT})")
    endif()
    # The Gradle-supplied variant is authoritative independently of the
    # effective label above (defense in depth against a stale cache).
    if(dfroot_variant_lower STREQUAL "release")
        message(FATAL_ERROR
            "DFROOT_SKIP_KO_HASH_CHECK is forbidden for the Android release "
            "variant (DFROOT_ANDROID_VARIANT=${DFROOT_ANDROID_VARIANT})")
    endif()
    message(WARNING
        "DFROOT_SKIP_KO_HASH_CHECK=ON is DEPRECATED and UNSUPPORTED: only the "
        "source-tree .ko SHA-256 pin check is DISABLED for a local experiment. "
        "It cannot bypass the ALWAYS-RUN final packaged-KO byte gate, cannot "
        "affect any distributable artifact, and an unknown blob still fails "
        "the build there. Never use it for CI, release or publishing builds. "
        "Reason: '${DFROOT_SKIP_REASON}'")
endif()

# Always-running target touches a real file that the stamp rule depends on;
# this forces verification on every build even if the blob is replaced with
# its original mtime. verify_ko_sha256.cmake rewrites the stamp only when its
# content changes, so exp.c is not recompiled when the hash is unchanged.
set(KO_FORCE_FILE "${CMAKE_CURRENT_BINARY_DIR}/dfroot-ko.verify.force")
add_custom_target(ko_sha256_force
    COMMAND "${CMAKE_COMMAND}" -E touch "${KO_FORCE_FILE}"
    BYPRODUCTS "${KO_FORCE_FILE}"
    COMMENT "Scheduling dfroot.ko SHA-256 re-check"
)

set(KO_VERIFY_DEPENDS
    "${KO_VERIFY_SCRIPT}"
    "${CMAKE_CURRENT_LIST_FILE}"   # re-verify if the pin itself is edited
    "${KO_FORCE_FILE}"
    ko_sha256_force                # also gives target-level ordering
)
if(NOT DFROOT_SKIP_KO_HASH_CHECK)
    list(APPEND KO_VERIFY_DEPENDS ${KO_FILES})
endif()

add_custom_command(
    OUTPUT "${KO_VERIFY_STAMP}"
    COMMAND "${CMAKE_COMMAND}"
            "-DKO_FILE=${KO_FILE}"
            "-DKO_EXPECTED_SHA256S=${KO_EXPECTED_SHA256S}"
            "-DKO_STAMP=${KO_VERIFY_STAMP}"
            "-DKO_VERIFY_MANIFEST=${KO_VERIFY_MANIFEST}"
            "-DDFROOT_BUILD_TYPE=${DFROOT_EFFECTIVE_BUILD_TYPE}"
            "-DDFROOT_ANDROID_VARIANT=${DFROOT_ANDROID_VARIANT}"
            "-DDFROOT_SKIP=${DFROOT_SKIP_KO_HASH_CHECK}"
            "-DDFROOT_SKIP_REASON=${DFROOT_SKIP_REASON}"
            -P "${KO_VERIFY_SCRIPT}"
    DEPENDS ${KO_VERIFY_DEPENDS}
    BYPRODUCTS "${KO_VERIFY_MANIFEST}"
    COMMENT "Verifying pinned dfroot.ko SHA-256"
    VERBATIM
)
add_custom_target(ko_sha256_check DEPENDS "${KO_VERIFY_STAMP}")

# Bundled userspace loader gate (P1-08 rebuild integration). The two binaries

# above; no KO pin or gate is changed here. There is deliberately no skip
# option: verify_loader_hashes.cmake removes any previous verified manifest at

# never survive a failed re-check. The force target makes this run on EVERY

set(LOADER_KSUD_FILE "${JNI_DIR}/../jniLibs/arm64-v8a/libksud.so")

set(PROVENANCE_FORCE_FILE "${CMAKE_CURRENT_BINARY_DIR}/dfroot-provenance.force")

# Final packaged-byte gate as a NON-CYCLIC two-stage artifact chain:
#   1. the raw link target `dfroot` emits an UNVERIFIED libdfroot.so into the
#      staging directory (never into AGP's packaging directory);
#   2. a custom command verifies that staged .so with verify_packaged_ko.cmake
#      and only then copies the identical bytes to the package path; a force
#      file touched by an always-running target re-runs the check on EVERY
#      build, even when the link itself is up to date;
#   3. the ALL target `dfroot_packaged` depends on the VERIFIED artifact, and
#      app/build.gradle.kts names that target in externalNativeBuild so AGP's
#      own build invocation traverses the same chain.
# No skip option: a failure fails the build and invalidates the manifest and
# verification work; the first command below also removes any previously
# verified copy at the package path before the re-check, so a stale artifact
# cannot survive a failed re-verification.
# CI repeats this gate on the bytes extracted from EVERY APK it uploads.
set(PACKAGED_KO_FORCE_FILE "${CMAKE_CURRENT_BINARY_DIR}/dfroot-packaged-ko.verify.force")
add_custom_target(dfroot_packaged_ko_force
    COMMAND "${CMAKE_COMMAND}" -E touch "${PACKAGED_KO_FORCE_FILE}"
    BYPRODUCTS "${PACKAGED_KO_FORCE_FILE}"
    COMMENT "Scheduling packaged-KO byte re-check"
)
add_custom_command(
    OUTPUT "${PACKAGED_KO_VERIFIED_SO}"
    # Drop any previously verified package-path copy before re-verifying.
    COMMAND "${CMAKE_COMMAND}" -E rm -f "${PACKAGED_KO_VERIFIED_SO}"
    COMMAND "${CMAKE_COMMAND}"
            "-DPACKAGED_SO=${PACKAGED_KO_RAW_SO}"
            "-DKO_OBJCOPY=${CMAKE_OBJCOPY}"
            "-DKO_EXPECTED_SHA256S=${KO_EXPECTED_SHA256S}"
            "-DKO_EXPECTED_SIZE=${KO_EXPECTED_SIZES}"
            "-DKO_MANIFEST=${PACKAGED_KO_MANIFEST}"
            "-DPACKAGED_VERIFIED_SO=${PACKAGED_KO_VERIFIED_SO}"
            "-DDFROOT_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/../../.."
            -P "${PACKAGED_KO_VERIFY_SCRIPT}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${PACKAGED_KO_RAW_SO}" "${PACKAGED_KO_VERIFIED_SO}"
    DEPENDS dfroot
            "${PACKAGED_KO_VERIFY_SCRIPT}"
            "${CMAKE_CURRENT_LIST_FILE}"   # re-verify if the pin itself is edited
            "${PACKAGED_KO_FORCE_FILE}"
            dfroot_packaged_ko_force       # target-level ordering: force runs first
    BYPRODUCTS "${PACKAGED_KO_MANIFEST}"
    COMMENT "Verifying packaged .dfroot_ko bytes in libdfroot.so"
    VERBATIM
)
add_custom_target(dfroot_packaged ALL DEPENDS "${PACKAGED_KO_VERIFIED_SO}")

# Master is the SINGLE status protocol. PR #2's run_status.c and
# loader_stage.c are intentionally NOT linked into either binary: duplicate
# symbols and contradictory outcomes. run_config_codec.c is the shared strict
# Java/native run-config codec (not a status protocol) retained from PR #2.
add_executable(bootstrap bootstrap.c bootstrap_stage.c status_protocol.c final_status_publish.c proc_scan_policy.c run_config_codec.c)
set_target_properties(bootstrap PROPERTIES
    PREFIX "lib"
    SUFFIX ".so"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
target_link_options(bootstrap PRIVATE -Wl,-z,max-page-size=16384)
# libbootstrap.so is a SEPARATE native artifact: ExploitRunner stages it from
# nativeLibraryDir at runtime, and it is not otherwise in dfroot_packaged's
# dependency graph. app/build.gradle.kts names both targets for AGP; this
# dependency is belt-and-braces so the packaged target cannot be built without
# the runtime-required bootstrap library. No cycle: bootstrap has no dependency
# on the dfroot link or verification chain.
add_dependencies(dfroot_packaged bootstrap)
# Also traverse the loader gate when AGP builds the named packaging targets
# instead of the default ALL target: no distributable libdfroot.so without a
# successful loader hash/size verification.
add_dependencies(dfroot_packaged loader_hash_check)
